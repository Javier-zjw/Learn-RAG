# ---- LLM（生成、LLM 精排、HyDE 改写、LLM 裁判都用它）-----------------------
LLM_BASE_URL=https://api.deepseek.com/v1
LLM_API_KEY=sk-your-key-here
LLM_MODEL=deepseek-chat

# ---- Embedding（与 LLM 可以是不同服务商）------------------------------------
EMBEDDING_BASE_URL=https://api.openai.com/v1
EMBEDDING_API_KEY=sk-your-key-here
EMBEDDING_MODEL=text-embedding-3-small
# 必须与模型实际输出维度一致；换模型后记得删 .cache/ 并重建向量库
EMBEDDING_DIM=1536
# 单次请求最多几条文本。各家上限不同（通义约 10、部分服务 25~64、OpenAI 2048）。
# 超了会自动减半重试并提示，探测出来的值写回这里即可。
EMBEDDING_BATCH_SIZE=32

# ---- Rerank（与 embedding 是两个独立模型，别共用 URL）------------------------
RERANK_BASE_URL=https://api.siliconflow.cn/v1
RERANK_API_KEY=sk-your-key-here
RERANK_MODEL=BAAI/bge-reranker-v2-m3
